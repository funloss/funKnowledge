table 
length(rows) as "笔记数",
sum(rows.file.size) as "字数",
max(rows.file.mtime) as "最近修改时间"
FROM "读书"   
where file.folder != "读书"  
group by split(replace(file.folder, "读书/",""), "/")[0] as "一级分类"
SORT length(rows)  DESC
